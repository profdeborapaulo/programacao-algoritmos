const prompt = require("prompt-sync")();

let resposta = "s";

while (resposta == "s") {

    let numero = Number(prompt("Digite um número: "));

    for (let i = 1; i <= 10; i++) {
        console.log(numero + " x " + i + " = " + (numero * i));
    }

    resposta = prompt("Deseja ver a tabuada de outro número??? (s/n): ").toLowerCase();

    while (resposta != "s" && resposta != "n") {
        resposta = prompt("Porfavor, digite apenas s ou n! (s: Sim, n: Não): ").toLowerCase();
    }
}

console.log("Encerrando o programa, até mais! obrigado por utilizar.");
