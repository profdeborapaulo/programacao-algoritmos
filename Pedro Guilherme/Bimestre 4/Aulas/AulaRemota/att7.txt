const prompt = require("prompt-sync")();

let contador = 0;

for (let i = 1; i <= 20; i++) {
    let numero = Number(prompt("Digite o " + i + "º número: "));

    if (numero >= 0 && numero <= 100) {
        contador++;
    }
}

console.log("Quantidade de números entre 0 e 100: " + contador);
